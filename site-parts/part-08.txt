function citTileXY(i){ const [r,c]=citRC(i); return [c*citTs, r*citTs]; }
function citNb(i){
  const [r,c]=citRC(i), out=[];
  [[r-1,c],[r+1,c],[r,c-1],[r,c+1]].forEach(([nr,nc])=>{
    if(nr>=0&&nc>=0&&nr<CITY_CFG.ROWS&&nc<CITY_CFG.COLS) out.push(citIdx(nr,nc));
  });
  return out;
}
const citWalkableTile=t=>t==='R'||t==='A';
function citWeekNum(){ return Math.floor((((cit&&cit.day)||1)-1)/7); }
function citDayOfWeek(){ return ((((cit&&cit.day)||1)-1)%7+7)%7; }

/* ---------- farm geometry ---------- */
function citFarmCells(center){
  if (center==null) return null;
  const [r,c]=citRC(center);
  if (r<1||c<1||r>=CITY_CFG.ROWS-1||c>=CITY_CFG.COLS-1) return null;
  const out=[];
  for(let dr=-1;dr<=1;dr++) for(let dc=-1;dc<=1;dc++) out.push(citIdx(r+dr,c+dc));
  return out;
}
function citFarmCenters(){
  const out=[]; cit.g.forEach((t,i)=>{ if(t==='M') out.push(i); }); return out;
}
function citFarmByCell(i){
  const [r,c]=citRC(i);
  for(let dr=-1;dr<=1;dr++) for(let dc=-1;dc<=1;dc++){
    const rr=r+dr, cc=c+dc;
    if(rr<0||cc<0||rr>=CITY_CFG.ROWS||cc>=CITY_CFG.COLS) continue;
    const j=citIdx(rr,cc);
    if(cit.g[j]==='M'){
      const cells=citFarmCells(j);
      if(cells && cells.includes(i)) return j;
    }
  }
  return null;
}
function citCleanPartialFarms(){
  const valid=new Set();
  (cit.g||[]).forEach((t,i)=>{
    if(t!=='M') return;
    const cells=citFarmCells(i);
    if(!cells || cells.some(j=>cit.g[j]!=='M'&&cit.g[j]!=='Q')){ cit.g[i]=0; return; }
    cells.forEach(j=>valid.add(j));
  });
  (cit.g||[]).forEach((t,i)=>{ if((t==='M'||t==='Q')&&!valid.has(i)) cit.g[i]=0; });
}
function citFarmNearRoad(center){
  const cells=citFarmCells(center); if(!cells) return false;
  return cells.some(i=>citNb(i).some(j=>citWalkableTile(cit.g[j])));
}
function citFarmRoads(center,act){
  const cells=citFarmCells(center)||[], out=new Set();
  cells.forEach(i=>citNb(i).forEach(j=>{ if(act.has(j)) out.add(j); }));
  return [...out];
}
function citFarmEmoji(i){
  const center=citFarmByCell(i); if(center==null) return '';
  const cells=citFarmCells(center)||[];
  const pos=cells.indexOf(i);
  return CITY_CFG.FARM_DECOR[(pos<0?0:pos)%CITY_CFG.FARM_DECOR.length];
}
function citFarmVacantCenters(){
  const occ=new Set(cit.ppl.map(p=>p.h));
  return citFarmCenters().filter(i=>!occ.has(i));
}

  if(!p.kidEl){
    p.kidEl=citPersonEl(p.kidFace||CITY_CFG.KID_FACES[0]); p.kidEl.classList.add('kid');
    p.kidEl.style.fontSize=(citTs*CITY_CFG.CITIZEN_SCALE*CITY_CFG.KID_SCALE)+'px';
  }
  citSetFace(p.kidEl,p.kidFace||CITY_CFG.KID_FACES[0]); citPlacePerson(p.kidEl,atIdx,false); p.kidAt=atIdx;
}
function citMergeFamily(p){
  if(!p.el)return;
  if(!p.fam){ if(p.at===p.h) citHomePose(p.el,null); return; }
  if(!p.kidEl) citSpawnKid(p,p.h);
  if(p.at===p.h && p.kidAt===p.h){
    citSetFace(p.el,p.face); citSetFace(p.kidEl,p.kidFace||CITY_CFG.KID_FACES[0]);
    citPlacePerson(p.el,p.h,false); citPlacePerson(p.kidEl,p.h,false);
    citHomePose(p.el,'parent'); citHomePose(p.kidEl,'kid');
  }
}

/* ---------- citizen metadata / preferences ---------- */
const CIT_JOB_PREFS=['O','M','S'];
function citRandomPreference(){ return CIT_JOB_PREFS[Math.floor(Math.random()*CIT_JOB_PREFS.length)]; }
function citPreferenceForId(id){
  /* Stable fallback only for legacy people that have no stored preference. */
  const n=Math.abs(((Number(id)||0)*1103515245+12345)>>>0);
  return CIT_JOB_PREFS[n%CIT_JOB_PREFS.length];
}
function citOffDaysForId(id){
  let a=(id*3+1)%5, b=(id*7+3)%5; if(a===b) b=(b+2)%5; return [a,b].sort();
}
function citEnsurePersonMeta(p,n){
  if(p.id==null) p.id=cit.personSeq++;
  if(!p.pref) p.pref=citPreferenceForId(p.id);
  if(!Array.isArray(p.offDays)||p.offDays.length!==2) p.offDays=citOffDaysForId(p.id);
  if(!p.homeKind) p.homeKind=(cit.g[p.h]==='M'?'M':'H');
  if(p.homeKind==='M') p.farmHome=p.h; else p.farmHome=null;
}
function citNewcomerTemplate(){
  const seq=cit.arr||0, id=cit.personSeq||0;
  const fam=(seq%CITY_CFG.FAMILY_EVERY)===(CITY_CFG.FAMILY_EVERY-1);
  return { id, face:CITY_CFG.FACES[seq%CITY_CFG.FACES.length], fam,
    kidFace:fam?CITY_CFG.KID_FACES[seq%CITY_CFG.KID_FACES.length]:null,
    pref:citRandomPreference(), offDays:citOffDaysForId(id) };
}
