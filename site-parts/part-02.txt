function tone(f, when, dur, type, vol){
  const c = ac(); if (!c || S.muted) return;
  const o = c.createOscillator(), g = c.createGain();
  o.type = type||'sine'; o.frequency.setValueAtTime(f, c.currentTime+when);
  g.gain.setValueAtTime(0.0001, c.currentTime+when);
  g.gain.linearRampToValueAtTime(vol||0.18, c.currentTime+when+0.015);
  g.gain.exponentialRampToValueAtTime(0.0001, c.currentTime+when+dur);
  o.connect(g); g.connect(c.destination);
  o.start(c.currentTime+when); o.stop(c.currentTime+when+dur+0.05);
}

  g.gain.setValueAtTime(vol||0.16,c.currentTime); g.gain.exponentialRampToValueAtTime(0.0001,c.currentTime+dur);

function confetti(n){
  const colors=['#ff5252','#ffd740','#69f0ae','#40c4ff','#e040fb','#ffab40'];
  for(let i=0;i<(n||90);i++) parts.push({ x:Math.random()*fx.width, y:-20-Math.random()*fx.height*0.3,
    vx:(Math.random()-0.5)*2.4, vy:2+Math.random()*3.5, r:4+Math.random()*6, c:colors[i%colors.length], a:Math.random()*6.28, va:(Math.random()-0.5)*0.3 });
  if (!fxRunning){ fxRunning=true; requestAnimationFrame(fxTick); }
}
function fxTick(){
  fctx.clearRect(0,0,fx.width,fx.height);
  parts.forEach(p=>{ p.x+=p.vx; p.y+=p.vy; p.a+=p.va; fctx.save(); fctx.translate(p.x,p.y); fctx.rotate(p.a); fctx.fillStyle=p.c; fctx.fillRect(-p.r/2,-p.r/2,p.r,p.r*0.6); fctx.restore(); });
  parts = parts.filter(p=>p.y < fx.height+30);
  if (parts.length){ requestAnimationFrame(fxTick); } else { fxRunning=false; fctx.clearRect(0,0,fx.width,fx.height); }
}
