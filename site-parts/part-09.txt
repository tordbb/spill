/* ---------- network/path finding ---------- */
function citActive(){
  const act=new Set(),q=[];
  cit.g.forEach((t,i)=>{
    if(citWalkableTile(t)){
      const [r,c]=citRC(i);
      if(r===0||c===0||r===CITY_CFG.ROWS-1||c===CITY_CFG.COLS-1){act.add(i);q.push(i);}
    }
  });
  while(q.length){ citNb(q.pop()).forEach(j=>{ if(citWalkableTile(cit.g[j])&&!act.has(j)){act.add(j);q.push(j);} }); }
  return act;
}
function citAdjRoads(i,act){
  const direct=citNb(i).filter(j=>act.has(j)); if(direct.length)return direct;
  const farm=citFarmByCell(i); if(farm!=null)return citFarmRoads(farm,act); return [];
}
function citConnected(i,act){ return act.has(i)||citAdjRoads(i,act).length>0; }
function citRoute(a,b,act){
  if(a===b)return[a];
  const starts=act.has(a)?[a]:citAdjRoads(a,act), goals=new Set(act.has(b)?[b]:citAdjRoads(b,act));
  if(!starts.length||!goals.size)return null;
  const prev={},seen=new Set(starts),q=[...starts]; let found=null;
  while(q.length){
    const cur=q.shift(); if(goals.has(cur)){found=cur;break;}
    citNb(cur).forEach(j=>{if(act.has(j)&&!seen.has(j)){seen.add(j);prev[j]=cur;q.push(j);}});
  }
  if(found==null)return null;
  const path=[found]; let k=found; while(prev[k]!=null){k=prev[k];path.unshift(k);}
  if(!act.has(a))path.unshift(a); if(!act.has(b))path.push(b); return path;
}
function citRoadPath(homeIdx,act){
  const starts=citAdjRoads(homeIdx,act); if(!starts.length)return null;
  const prev={},seen=new Set(starts),q=[...starts]; let edge=null;
  while(q.length){
    const cur=q.shift(),[r,c]=citRC(cur);
    if(r===0||c===0||r===CITY_CFG.ROWS-1||c===CITY_CFG.COLS-1){edge=cur;break;}
    citNb(cur).forEach(j=>{if(act.has(j)&&!seen.has(j)){seen.add(j);prev[j]=cur;q.push(j);}});
  }
  if(edge==null)return null;
  const path=[edge]; let k=edge; while(prev[k]!=null){k=prev[k];path.push(k);} path.push(homeIdx); return path;
}
function citNearSchool(homeIdx,act){
  let best=null,bd=Infinity;
  cit.g.forEach((t,i)=>{
    if(t!=='K'||!citConnected(i,act))return; const route=citRoute(homeIdx,i,act); if(!route)return;
    const steps=route.length-1; if(steps<CITY_CFG.SCHOOL_MAX_STEPS&&steps<bd){bd=steps;best=i;}
  }); return best;
}

async function citWalkW(w,toIdx,act){
  if(!w.el){w.at=toIdx;return true;} citMovingPose(w.el);
  const from=w.at!=null?w.at:toIdx; if(from===toIdx)return true;

  if(!path){
    /* Never glide through buildings/grass as a fallback: all city travel must stay on the road network. */
    w.el.style.transition=''; return false;
  }
  w.el.style.transition=`left ${CITY_CFG.WALK_STEP_MS+70}ms linear, top ${CITY_CFG.WALK_STEP_MS+70}ms linear`;
  for(let k=1;k<path.length;k++){citPlacePerson(w.el,path[k],true);await citSleep(CITY_CFG.WALK_STEP_MS);}
  w.el.style.transition=''; w.at=toIdx; return true;
}

function citFarmWorkersAvailable(act,reachable){
  const farms=citFarmCenters().map(i=>({i,used:0}));
  const workers=[],idle=[];
  cit.ppl.filter(p=>p.pref==='M').forEach(p=>{
    let f=null;
    if(cit.g[p.h]==='M') f=farms.find(x=>x.i===p.h&&x.used<CITY_CFG.FARM_WORK_CAP);
    if(!f) f=farms.find(x=>x.used<CITY_CFG.FARM_WORK_CAP&&reachable(p.h,x.i));
    if(f){f.used++;p.kind='M';p.dest=f.i;workers.push(p);} else idle.push(p);
  });
  return {workers,idle,farms};
}
async function citFarmActivity(p,center,hops){
  const cells=citFarmCells(center)||[]; if(!cells.length)return;
  const targets=cells.filter(i=>i!==center);
  for(let n=0;n<(hops||2)&&targets.length;n++){
    const to=targets[(p.id+n*3+cit.day)%targets.length];await citHop(p.el,to,CITY_CFG.FARM_HOP_MS);p.at=to;
  }
  await citHop(p.el,center,CITY_CFG.FARM_HOP_MS);p.at=center;
}
function citShopDeliveryTarget(center,act,used){
  const week=citWeekNum(),list=[];
  cit.g.forEach((t,i)=>{
    if(t!=='S'||used.has(i)||!citConnected(i,act)||!citRoute(center,i,act))return;
    const last=Number(cit.shopStock[i]); if(last<week)list.push({i,last:Number.isFinite(last)?last:-999});
  });
  list.sort((a,b)=>a.last-b.last||a.i-b.i);return list.length?list[0].i:null;
}
async function citFarmDeliveries(farmWorkers,act){
  const used=new Set(), week=citWeekNum();
  for(const p of farmWorkers){
    /* One farm makes at most one boutique delivery per week. */
    if(citFarmDeliveredThisWeek(p.dest,week)) continue;
    const target=citShopDeliveryTarget(p.dest,act,used);if(target==null)continue;used.add(target);
    citClearThought(p.el); citSetFace(p.el,'🚙');
    try {
      const arrived=await citWalkW(p,target,act); if(!arrived)continue;
      citMarkShopStocked(target,week); citMarkFarmDelivered(p.dest,week); citRenderTiles();
      await citEarn(target,CITY_CFG.FARM_DELIVERY_INCOME);
      await citWalkW(p,p.dest,act); await citFarmActivity(p,p.dest,1);
    } finally {
      /* A failed/aborted delivery can never leave a car/truck face stuck on a shop or road. */
      citSetFace(p.el,p.face);
    }
  }
}
