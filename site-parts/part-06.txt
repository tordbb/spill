  patFails++;

function patBFS(){
  const rows=patGrid.length, cols=patGrid[0].length;
  const prev={}, seen=new Set([patStart.join()]), q=[patStart];
  while(q.length){
    const cur=q.shift();
    if (cur[0]===patFish[0]&&cur[1]===patFish[1]){
      const cells=[]; let k=cur.join();
      while(prev[k]){ cells.unshift(k.split(',').map(Number)); k=prev[k]; }
      return cells;
    }
    for (const dk in DIRS){
      const d=DIRS[dk], nr=cur[0]+d[0], nc=cur[1]+d[1];
      if (nr<0||nc<0||nr>=rows||nc>=cols||patGrid[nr][nc]==='#') continue;
      const key=nr+','+nc; if (seen.has(key)) continue; seen.add(key);
      prev[key]=cur.join(); q.push([nr,nc]);
    }
  }
  return [];
}
function patShowTrail(){
  let cells;
  if ((patLevel.robots||patLevel.dogs) && patLevel.sol) cells = patSimulate(patLevel.sol).catCells.slice(1);
  else cells = patBFS();
  if (!cells.length) return;
  const g=$('#pat-grid');
  const pawEls=[];
  cells.forEach((rc,i)=>{
    if (rc[0]===patFish[0]&&rc[1]===patFish[1]) return;
    if (rc[0]===patStart[0]&&rc[1]===patStart[1]) return;
    const p=document.createElement('div'); p.className='paw'; p.textContent='🐾';
    p.style.left=(10+rc[1]*(patTs+4))+'px'; p.style.top=(10+rc[0]*(patTs+4))+'px';
    p.style.opacity=0; p.style.transition='opacity .3s';
    g.appendChild(p); pawEls.push(p);
    setTimeout(()=>{ p.style.opacity=0.8; }, 300+i*220);
  });
  setTimeout(()=>{ pawEls.forEach(p=>p.remove()); }, 5600);
}

let walGlowEl=null;
function walCellFromEvent(e){
  const r=$('#wal-grid').getBoundingClientRect();
  if (e.clientX<r.left||e.clientX>r.right||e.clientY<r.top||e.clientY>r.bottom) return null;
  const c=Math.floor((e.clientX-r.left)/walWs), rr=Math.floor((e.clientY-r.top)/walWs);
  if (rr<0||c<0||rr>=walMap.length||c>=walMap[0].length) return null;
  return [rr,c];
}
function walCellFree(r,c){
  const ch=walMap[r][c];
  return ch!=='#' && ch!=='H' && !walPlaced[r+','+c] && !(r===walStart[0]&&c===walStart[1]);
}
function walToolDrag(el, tool, idx){
  el.addEventListener('pointerdown', e=>{
    if (walRun) return;
    e.preventDefault();
    el.setPointerCapture(e.pointerId);
    el.style.opacity=0.3;
    startGhost(TOOL_EMO[tool], e.clientX, e.clientY); sfx('pickup');
    const mv=ev=>{
      moveGhost(ev.clientX, ev.clientY);
      const cell=walCellFromEvent(ev);
      if (walGlowEl){ walGlowEl.remove(); walGlowEl=null; }
      if (cell && walCellFree(cell[0],cell[1])) walGlowEl=walTileAt(cell[0],cell[1],'cellglow','');
    };
    const up=ev=>{
      el.removeEventListener('pointermove',mv); el.removeEventListener('pointerup',up); el.removeEventListener('pointercancel',up);
      endGhost(); el.style.opacity=1;
      if (walGlowEl){ walGlowEl.remove(); walGlowEl=null; }
      const cell=walCellFromEvent(ev);
      if (cell && walCellFree(cell[0],cell[1])){
        walPlaced[cell[0]+','+cell[1]]=tool;
        walTools.splice(idx,1);
        sfx('drop');
        walRenderGrid(); walRenderTools();
      } else sfx('click');
    };
    el.addEventListener('pointermove',mv); el.addEventListener('pointerup',up); el.addEventListener('pointercancel',up);
  });
}
/* placed pieces can be dragged to a new spot, dragged off the grid to return them,
   or simply tapped to send them back to the toolbox */
function walPlacedDrag(el, key){
  el.addEventListener('pointerdown', e=>{
    if (walRun) return;
    e.preventDefault(); el.setPointerCapture(e.pointerId);
    const tool=walPlaced[key];
    const sx=e.clientX, sy=e.clientY;
    let moved=false;
    const mv=ev=>{
      if (!moved && Math.hypot(ev.clientX-sx, ev.clientY-sy)>8){
        moved=true; el.style.opacity=0.25;
        startGhost(TOOL_EMO[tool], ev.clientX, ev.clientY); sfx('pickup');
      }
      if (!moved) return;
      moveGhost(ev.clientX, ev.clientY);
      const cell=walCellFromEvent(ev);
      if (walGlowEl){ walGlowEl.remove(); walGlowEl=null; }
      if (cell && (walCellFree(cell[0],cell[1]) || cell.join(',')===key)) walGlowEl=walTileAt(cell[0],cell[1],'cellglow','');
    };
    const up=ev=>{
      el.removeEventListener('pointermove',mv); el.removeEventListener('pointerup',up); el.removeEventListener('pointercancel',up);
      if (walGlowEl){ walGlowEl.remove(); walGlowEl=null; }
      endGhost(); el.style.opacity=1;
      if (!moved){ // tap: back to toolbox
        sfx('pop'); walTools.push(tool); delete walPlaced[key];
      } else {
        const cell=walCellFromEvent(ev);
        if (cell && walCellFree(cell[0],cell[1])){ delete walPlaced[key]; walPlaced[cell[0]+','+cell[1]]=tool; sfx('drop'); }
        else if (!cell){ walTools.push(tool); delete walPlaced[key]; sfx('pop'); }
        else sfx('click');
      }
      walRenderGrid(); walRenderTools();
    };
    el.addEventListener('pointermove',mv); el.addEventListener('pointerup',up); el.addEventListener('pointercancel',up);
  });
}

function walJumpCritter(fromPos, peakRow, toPos, ms, done){
  const cr=$('#wal-critter'); if (!cr){ done(); return; }
  const sx=fromPos[1]*walWs, sy=fromPos[0]*walWs;
  const mx=((fromPos[1]+toPos[1])/2)*walWs, my=peakRow*walWs;
  const ex=toPos[1]*walWs, ey=toPos[0]*walWs;
  const face=`scaleX(${walDir})`;
  cr.classList.remove('fall');
  cr.style.transition='none';
  const frames=[
    { transform:`translate(${sx}px, ${sy}px) ${face}`, offset:0 },
    { transform:`translate(${mx}px, ${my}px) ${face}`, offset:.5 },
    { transform:`translate(${ex}px, ${ey}px) ${face}`, offset:1 }
  ];
  if (cr.animate){
    const a=cr.animate(frames,{duration:ms,easing:'linear',fill:'forwards'});
    a.onfinish=()=>{
      cr.style.transform=frames[2].transform;
      a.cancel();
      cr.style.transition='';
      done();
    };
  } else {
