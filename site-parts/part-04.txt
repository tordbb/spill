function creKey(p){ return p.shape+'|'+p.dims.join(',')+'|'+p.color; }
function svgEl(tag, attrs, parent){

function shapeInto(g, p, fill){
  const common={ fill: fill||p.color, stroke:'#00000026', 'stroke-width':0.6 };
  if (p.shape==='c') svgEl('circle', Object.assign({r:p.dims[0]}, common), g);
  else if (p.shape==='r') svgEl('rect', Object.assign({x:-p.dims[0]/2, y:-p.dims[1]/2, width:p.dims[0], height:p.dims[1], rx:1.6}, common), g);
  else if (p.shape==='t'){ const s=p.dims[0];
    svgEl('polygon', Object.assign({points:`0,${-s} ${s*0.87},${s*0.5} ${-s*0.87},${s*0.5}`}, common), g); }
}

function creAttachDrag(cp, lv){
  cp.el.addEventListener('pointerdown', e=>{
    if (cp.placed && !creFree) return;
    e.preventDefault();
    cp.el.parentNode.appendChild(cp.el); // to front (before capture: re-inserting cancels capture)
    cp.el.setPointerCapture(e.pointerId);
    const [sx,sy]=creSvgPos(e); const ox=cp.x-sx, oy=cp.y-sy;
    let moved=false;
    sfx('pickup');
    if (!creFree){
      creParts.forEach(o=>{ if (!o.filled && creKey(o.def)===creKey(cp.def)) o.slotEl.classList.add('glow'); });
    }
    const mv=ev=>{ const [x,y]=creSvgPos(ev); cp.x=x+ox; cp.y=y+oy; moved=true;
      creTransform(cp.el, cp.x, cp.y, cp.rot!=null?cp.rot:cp.def.rot); };
    const up=ev=>{
      cp.el.removeEventListener('pointermove',mv); cp.el.removeEventListener('pointerup',up); cp.el.removeEventListener('pointercancel',up);
      $$('.slot').forEach(s=>s.classList.remove('glow'));
      if (creFree){ creFreeDrop(cp, moved); return; }
      // find nearest matching free slot
      let best=null, bd=11;
      creParts.forEach(o=>{ if (!o.filled && creKey(o.def)===creKey(cp.def)){
        const d=Math.hypot(o.def.x-cp.x, o.def.y-cp.y); if (d<bd){ bd=d; best=o; } } });
      if (best){
        best.filled=true; cp.placed=true;
        cp.x=best.def.x; cp.y=best.def.y;
        creTransform(cp.el, cp.x, cp.y, best.def.rot);
        creRestack(cp, best.idx);
        best.slotEl.remove();
        sfx('snap');
        creCheckDone(lv);
      } else {
        cp.x=cp.hx; cp.y=cp.hy;
        creTransform(cp.el, cp.x, cp.y, cp.def.rot);
        if (moved) sfx('drop');
      }
    };
    cp.el.addEventListener('pointermove',mv); cp.el.addEventListener('pointerup',up); cp.el.addEventListener('pointercancel',up);
  });
}
function creRestack(cp, slotIdx){
  // keep authored part order (background parts first) among placed pieces
  cp.el.dataset.z = slotIdx;
  const live=$('#cre-live');
  const after=[...live.children].find(ch => ch!==cp.el && ch.dataset.z!==undefined && +ch.dataset.z > slotIdx);
  if (after) live.insertBefore(cp.el, after);
}
