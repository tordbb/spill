    // Fallback: two equal-duration straight segments, peaking halfway horizontally.
    cr.style.transition=`transform ${ms/2}ms linear`;
    requestAnimationFrame(()=>{
      cr.style.transform=frames[1].transform;
      setTimeout(()=>{
        cr.style.transform=frames[2].transform;
        setTimeout(()=>{ cr.style.transition=''; done(); }, ms/2);
      }, ms/2);
    });
  }
}
function walBounceLanding(r, nc, peakRow){
  // Preserve the old bounce reach: rise up to three rows, then land on the first
  // supported free cell in the forward column. Only the visual path changes.
  let lr=peakRow;
  while (lr<walMap.length && walSolid(lr,nc)) lr--;
  if (lr<0) lr=0;
  while (lr+1<walMap.length && !walSolid(lr+1,nc)) lr++;
  if (lr+1>=walMap.length) return null;
  return [lr,nc];
}
function walGo(){
  walRun=true; walDir=1; walPos=[...walStart];
  walMoveCritter(walPos, false);
  let ticks=0, lastKey='';
  const cr=$('#wal-critter');
  const tick=()=>{
    ticks++;
    if (ticks>60){ walFail(); return; }
    let [r,c]=walPos;
    if (r===walHome[0]&&c===walHome[1]){ walWin(); return; }
    // falling?
    if (!walSolid(r+1,c)){
      if (r+1>=walMap.length){ walFail(); return; }
      walPos=[r+1,c]; cr.classList.add('fall');
      walMoveCritter(walPos,true); sfx('step');
      setTimeout(tick,320); return;
    }
    cr.classList.remove('fall');
    const key=r+','+c;
    const eff=walPlaced[key];
    if (eff==='M' && lastKey!=='M'+key){
      lastKey='M'+key;
      const nc=c+walDir;
      if (nc<0||nc>=walMap[0].length){ walFail(); return; }
      const peakRow=Math.max(0,r-3);
      const landing=walBounceLanding(r,nc,peakRow);
      if (!landing){ walFail(); return; }
      const from=[r,c];
      walPos=[...landing];
      sfx('boing');
      // The jump is two straight, equally timed diagonals: apex exactly halfway
      // between the start and landing columns, then the mirrored descent.
      walJumpCritter(from, peakRow, landing, 720, ()=>setTimeout(tick,80));
      return;
    }
    if (eff==='F' && lastKey!=='F'+key){
      lastKey='F'+key;
      walDir=-walDir; sfx('pop');
      walMoveCritter(walPos,true);
      setTimeout(tick,360); return;
    }
    // walk
    const nc=c+walDir;
    if (nc<0||nc>=walMap[0].length){ walFail(); return; }
    if (walSolid(r,nc)){
      if (!walSolid(r-1,nc) && !walSolid(r-1,c)){ walPos=[r-1,nc]; } // step up
      else { walDir=-walDir; walMoveCritter(walPos,true); setTimeout(tick,360); return; }
    } else walPos=[r,nc];
    if (lastKey && walPos.join(',')!==key) lastKey='';
    walMoveCritter(walPos,true); sfx('step');
    setTimeout(tick,340);
  };
  setTimeout(tick,350);
}
function walFail(){
  walFails++;
  const cr=$('#wal-critter');
  if (cr){
    cr.style.transition='opacity .7s ease, transform .7s ease';
    cr.style.opacity='0';
  }
  slide(500,180,0.4,'sine',0.1);
  cheer();
  setTimeout(()=>{
    walDir=1; walPos=[...walStart];
    walRenderGrid();
    walRun=false;
  }, 850);
}
function walWin(){
  const cr=$('#wal-critter');
  cr.style.transition='transform .3s';
  cr.style.transform+=' scale(1.2)';
  walTileAt(Math.max(0,walHome[0]-1), walHome[1], '', '💕');
  setTimeout(()=>{
    walRun=false;
    award('wal', walCur, ()=>{ if (walCur+1<WAL_LEVELS.length) openWal(walCur+1); else goHome(); });
  }, 700);
}

function citResizeState(newCols,newRows){
  if (!cit){ CITY_CFG.COLS=newCols; CITY_CFG.ROWS=newRows; cit=citDefault(); S.city=cit; return; }
  const [oldCols,oldRows]=citInferDims(cit);
  if (oldCols===newCols && oldRows===newRows){ cit.cols=newCols; cit.rows=newRows; CITY_CFG.COLS=newCols; CITY_CFG.ROWS=newRows; return; }
  const oldG=cit.g.slice(), oldSt=Object.assign({},cit.st||{}), oldStock=Object.assign({},cit.shopStock||{}), oldFarmDelivery=Object.assign({},cit.farmDelivery||{});
  const ng=Array(newCols*newRows).fill(0), nst={}, nstock={}, nfd={};
  const rr=Math.min(oldRows,newRows), cc=Math.min(oldCols,newCols);
  for (let r=0;r<rr;r++) for (let c=0;c<cc;c++){
    const oi=r*oldCols+c, ni=r*newCols+c;
    ng[ni]=oldG[oi]||0;
    if (oldSt[oi]!=null) nst[ni]=oldSt[oi];
    if (oldStock[oi]!=null) nstock[ni]=oldStock[oi];
  }
  const mapIdx=i=>{
    if (i==null) return i;
    const r=Math.floor(i/oldCols), c=i%oldCols;
    return (r<newRows&&c<newCols)?r*newCols+c:null;
  };
  for(const [k,v] of Object.entries(oldFarmDelivery)){ const ni=mapIdx(Number(k)); if(ni!=null) nfd[ni]=v; }
  cit.ppl=(cit.ppl||[]).map(p=>{
    const q=Object.assign({},p);
    ['h','at','kidAt','dest','school','shop','spot','playStart'].forEach(k=>{ if(q[k]!=null) q[k]=mapIdx(q[k]); });
    return q;
  });
  cit.g=ng; cit.st=nst; cit.shopStock=nstock; cit.farmDelivery=nfd; cit.cols=newCols; cit.rows=newRows;
  CITY_CFG.COLS=newCols; CITY_CFG.ROWS=newRows;
  citCleanPartialFarms();
  cit.ppl=cit.ppl.filter(p=>p.h!=null && (cit.g[p.h]==='H'||cit.g[p.h]==='M'));
  citUndo=[];
}

function citValidateState(where){
  /* Development-time invariant checks. They do not change gameplay; they make future state bugs obvious in the console. */
  const problems=[];
  if(!cit||!Array.isArray(cit.g)) problems.push('missing grid');
  else if(cit.g.length!==CITY_CFG.COLS*CITY_CFG.ROWS) problems.push('grid-size mismatch');
  const ids=new Set(), homes=new Set();
  (cit&&cit.ppl||[]).forEach(p=>{
    if(ids.has(p.id))problems.push('duplicate person id '+p.id); ids.add(p.id);
    if(homes.has(p.h))problems.push('duplicate household home '+p.h); homes.add(p.h);
    if(p.h==null || (cit.g[p.h]!=='H'&&cit.g[p.h]!=='M')) problems.push('invalid home '+p.h);
  });
  if(problems.length) console.warn('[city invariant '+where+']',problems);
  return problems;
}
